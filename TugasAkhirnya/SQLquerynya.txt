Tabel Presentasi gross laba
SELECT 
  product_id,
  price,
  CASE 
    WHEN price <= 50000 THEN '10%'
    WHEN price > 50000 AND price <= 100000 THEN '15%'
    WHEN price > 100000 AND price <= 300000 THEN '20%'
    WHEN price > 300000 AND price <= 500000 THEN '25'
    WHEN price > 500000 THEN '35'
  END AS Persentase_gross_laba
FROM `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction` LIMIT 1000

Tabel Nett sales
SELECT 
  product_id,
  price,
  discount_percentage,
  CAST(ROUND(price - (price * discount_percentage)) AS INT64) AS `harga setelah diskon`
FROM `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction` LIMIT 1000

Tabel Net Profit
WITH perhitungan_awal AS (
  SELECT 
    product_id,
    price,
    discount_percentage,
    CAST(ROUND(price - (price * discount_percentage)) AS INT64) AS nett_sales,
    CASE 
      WHEN price <= 50000 THEN 0.10
      WHEN price > 50000 AND price <= 100000 THEN 0.15
      WHEN price > 100000 AND price <= 300000 THEN 0.20
      WHEN price > 300000 AND price <= 500000 THEN 0.25
      WHEN price > 500000 THEN 0.30
    END AS persentase_laba
  FROM `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction`
)

SELECT 
  product_id,
  price,
  discount_percentage,
  nett_sales,
  persentase_laba,
  CAST(ROUND(nett_sales * persentase_laba) AS INT64) AS nett_profit
FROM perhitungan_awal

- Menampilkan Tabel YOY Revenue Difference
WITH YearlyRevenue AS (
  SELECT 
    EXTRACT(YEAR FROM date) AS transaction_year,
    SUM(price) AS total_revenue -- Ganti 'price' dengan kolom yang merepresentasikan pendapatan (misal: nett_sales)
  FROM 
    `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction`
  GROUP BY 
    transaction_year
)
SELECT 
  transaction_year,
  total_revenue,
  LAG(total_revenue) OVER (ORDER BY transaction_year) AS previous_year_revenue,
  total_revenue - LAG(total_revenue) OVER (ORDER BY transaction_year) AS revenue_difference,
  ROUND(
    ((total_revenue - LAG(total_revenue) OVER (ORDER BY transaction_year)) / 
    NULLIF(LAG(total_revenue) OVER (ORDER BY transaction_year), 0)) * 100, 
  2) AS yoy_growth_percentage
FROM 
  YearlyRevenue
ORDER BY 
  transaction_year;

-Menampilkan tabel TopTenBranch Trx
SELECT 
  c.provinsi, 
  COUNT(t.transaction_id) AS jumlah_transaksi, 
  SUM(t.price) AS total_pendapatan 
FROM 
  `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction` t
JOIN 
  `careful-acumen-510622-j0.kimia_farma01.kf_kantor_cabang` c
ON 
  t.branch_id = c.branch_id 
GROUP BY 
  c.provinsi
ORDER BY 
  total_pendapatan DESC 
LIMIT 10;

SELECT  
  c.provinsi,
  CONCAT(ROUND(SUM(t.price) / 1000000, 2), ' Juta') AS total_profit
FROM 
  `careful-acumen-510622-j0.kimia_farma01.kf_kantor_cabang` AS c
JOIN
  `careful-acumen-510622-j0.kimia_farma01.kf_final_transaction` AS t
  ON c.branch_id = t.branch_id

  c.provinsi
ORDER BY
